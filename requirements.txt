# Installation Instructions:
# ========================
# 
# 1. Create virtual environment:
#    python -m venv venv
#    source venv/bin/activate  # On Windows: venv\Scripts\activate
#
# 2. Install PyTorch (visit https://pytorch.org for your specific system):
#    
#    # CPU-only (Linux/Mac):
#    pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cpu
#    
#    # CUDA 11.8 (Linux/Windows):
#    pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu118
#    
#    # CUDA 12.1 (Linux/Windows):
#    pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121
#    
#    # Mac (Apple Silicon):
#    pip install torch torchvision torchaudio
#
# 3. Install PyTorch Geometric:
#    
#    # For CUDA 11.8:
#    pip install torch-scatter torch-sparse torch-cluster torch-geometric \
#        -f https://data.pyg.org/whl/torch-2.0.0+cu118.html
#    
#    # For CUDA 12.1:
#    pip install torch-scatter torch-sparse torch-cluster torch-geometric \
#        -f https://data.pyg.org/whl/torch-2.0.0+cu121.html
#    
#    # For CPU:
#    pip install torch-scatter torch-sparse torch-cluster torch-geometric \
#        -f https://data.pyg.org/whl/torch-2.0.0+cpu.html
#
# 4. Install JAX:
#    
#    # CPU-only:
#    pip install jax jaxlib jax-md
#    
#    # GPU (CUDA 11.8):
#    pip install jax[cuda11_pip] jaxlib jax-md -f https://storage.googleapis.com/jax-releases/jax_cuda_releases.html
#    
#    # GPU (CUDA 12.1):
#    pip install jax[cuda12_pip] jaxlib jax-md -f https://storage.googleapis.com/jax-releases/jax_cuda_releases.html
#
# 5. Install remaining dependencies:
#    pip install -r requirements.txt
#
# Verification:
# =============
# Run this to verify installation:
#
#   python -c "import torch; print('PyTorch:', torch.__version__, '| CUDA:', torch.cuda.is_available())"
#   python -c "import jax; print('JAX:', jax.__version__, '| Devices:', jax.devices())"
#   python -c "import torch_geometric; print('PyG:', torch_geometric.__version__)"
#
